% Shared preamble for the CV (reference single-column, ATS-friendly style).
% Compiled with XeLaTeX so we can use Georgia directly (matches the .docx
% master, which also uses Georgia) instead of the default Latin Modern.
\usepackage{fontspec}
\setmainfont{Georgia}
\usepackage[a4paper,left=1.6cm,right=1.6cm,top=0.95cm,bottom=0.95cm]{geometry}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{xcolor}
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\linespread{1.05}

% Clickable links keep the same text colour (ATS-friendly, no blue soup).
\hypersetup{colorlinks=false, pdfborder={0 0 0}}

% ---- Section headings: centred, bold, thin rule under ----
% (No small caps: Georgia has no true small-caps glyphs, so \scshape would
% silently fall back to a substituted shape under fontspec. Bold title case
% reproduces the reference style just as well.)
\titleformat{\section}
  {\centering\large\bfseries}
  {}{0em}{}
  [\vspace{1pt}\hrule height 0.5pt\vspace{1pt}]
\titlespacing*{\section}{0pt}{7pt}{3pt}

% ---- One experience/education entry header (single line) ----
% #1 bold (org)  --  #2 italic (role/degree)  ........  #3 italic (dates)
\newcommand{\cvhead}[3]{%
  \textbf{#1} \textendash{} \textit{#2}\hfill\textit{#3}\par%
}

% ---- Compact bullet list ----
\newlist{cvitems}{itemize}{1}
\setlist[cvitems]{leftmargin=1.3em, labelsep=0.45em, itemsep=1.5pt,
  topsep=2pt, parsep=0pt, label=\textbullet}

% Spacing between entries
\newcommand{\entrygap}{\vspace{3.5pt}}
